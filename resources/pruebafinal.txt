#+ ************** Lote de Pruebas Completo ************** +#
init {
    a, b, c, d : Int
    e, f, g, h : Float
    i, j, k : String
    nombre : String
}

#+ 1. Inicializacion de Variables +#
write("1. Inicializando variables...\n")
a := 10
b := 3
c := 2
d := 5

e := 10.5
f := 3.2
g := 1.5
h := 2.0

i := "Hola"
j := "Mundo"

#+ 2. Operaciones Aritmeticas Basicas (Enteros y Flotantes sin Division por Cero) +#
write("2. Probando operaciones aritmeticas...\n")
a := b + c
write("Suma (3 + 2): ")
write(a)
write("\n")

a := b - c
write("Resta (3 - 2): ")
write(a)
write("\n")

a := b * c
write("Multiplicacion (3 * 2): ")
write(a)
write("\n")

a := b div c
write("Division entera (3 div 2): ")
write(a)
write("\n")

a := b mod c
write("Modulo entero (3 mod 2): ")
write(a)
write("\n")

#+ 3. Estructuras de Control (If / Else) +#
write("\n3. Probando condicionales (If / Else)...\n")
a := 10
b := 20
if (a < b) {
    write("a es menor que b (Correcto)\n")
} else {
    write("a no es menor que b (Incorrecto)\n")
}

if (a > b) {
    write("a es mayor que b (Incorrecto)\n")
} else {
    write("a no es mayor que b (Correcto)\n")
}

#+ 4. Bucles Simples (While que terminan) +#
write("\n4. Probando bucles while simples...\n")
a := 3
while (a > 0) {
    write("Iteracion simple a: ")
    write(a)
    write("\n")
    a := a - 1
}

#+ 5. Bucles Anidados +#
write("\n5. Probando bucles while anidados...\n")
a := 2
while (a > 0) {
    b := 2
    while (b > 0) {
        write("Anidado a y b: ")
        write(a)
        write(" ")
        write(b)
        write("\n")
        b := b - 1
    }
    a := a - 1
}

#+ 6. Ciclos Especiales (While in list) +#
write("\n6. Probando ciclos especiales...\n")
while a in [10, 20, 30] do
    write("Ciclo especial a: ")
    write(a)
    write("\n")
endwhile

#+ 7. Ciclos Especiales Anidados +#
write("\n7. Probando ciclos especiales anidados...\n")
while a in [1, 2] do
    while b in [100, 200] do
        write("Especial anidado a y b: ")
        write(a)
        write(" ")
        write(b)
        write("\n")
    endwhile
endwhile

#+ 8. Lecturas de Teclado con Prompts +#
write("\n8. Probando lecturas de teclado...\n")
write("Ingrese un numero entero para la variable a: ")
read(a)
write("El numero ingresado es: ")
write(a)
write("\n\n")

write("Pruebas completadas de manera exitosa!\n")
